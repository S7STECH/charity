// ───────────────────────────────────────────── // تصنيف الجمعيات حسب الإيرادات السنوية // ───────────────────────────────────────────── function classify() { let revenue = parseFloat( document.getElementById("revenue").value ); // التحقق من صحة المدخل if (isNaN(revenue) || revenue < 0) { document.getElementById("result").innerHTML = "⚠️ أدخل رقم صحيح"; return; } let result = ""; // شروط التصنيف if (revenue < 500_000) { result = "متناهية الصغر"; // أقل من 500,000 } else if (revenue <= 2_000_000) { result = "صغيرة"; // 500,000 – 2,000,000 } else if (revenue <= 8_000_000) { result = "متوسطة"; // 2,000,001 – 8,000,000 } else if (revenue <= 30_000_000) { result = "كبيرة"; // 8,000,001 – 30,000,000 } else { result = "متناهية الكبر"; // أكثر من 30,000,000 } // عرض النتيجة document.getElementById("result").innerHTML = "📊 التصنيف: " + result; }
